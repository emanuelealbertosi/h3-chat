cmake_minimum_required(VERSION 3.24)
project(h3_music LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
 add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:/utf-8>")
endif()
if(NOT AUDIOCPP_SOURCE_DIR)
  message(FATAL_ERROR "Set AUDIOCPP_SOURCE_DIR to the pinned, patched audio.cpp source")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(AUDIOCPP_BUILD_NATIVE_MODEL_MANAGER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory("${AUDIOCPP_SOURCE_DIR}" audio EXCLUDE_FROM_ALL)
add_executable(h3-music-worker worker.cpp)
target_include_directories(h3-music-worker PRIVATE ../vendor)
target_link_libraries(h3-music-worker PRIVATE engine_runtime ggml)
if(MSVC)
 target_compile_options(h3-music-worker PRIVATE /utf-8 /EHsc)
endif()
